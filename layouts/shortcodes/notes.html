{{- /* Named `type` only, matching the long-standing behavior. A positional argument — {{% notes "warning" %}} — is IGNORED and silently yields an info box; see "Pulumi Cloud availability markers" in AGENTS.md before changing this, because switching it on reclassifies existing content. */ -}} {{- $type := default "info" (.Get "type") -}} {{- /* The leading newline is load-bearing. `{{%` shortcode output is spliced back into the markdown source before it is rendered, so an indented call — several SCIM/SAML pages open one with four spaces of indentation inside a list — would put `
` at column 4 and Goldmark would render the whole callout as an indented code block. Starting the output on a fresh line keeps the div at column 0, which is what this shortcode has always emitted (the old template began with untrimmed `{{ $type := … }}` lines). Do not "clean up" by trimming this. */ -}} {{- "\n" -}} {{- /* Strip the body's common indentation before markdownify, the way Python's textwrap.dedent does. Authors reasonably indent a whole callout to line it up inside a list item, and four spaces of it would otherwise turn the body into a code block. Only the shared prefix goes, so nested lists and code blocks inside the callout keep their relative indentation. */ -}} {{- $inner := .Inner -}} {{- $indent := -1 -}} {{- range (split $inner "\n") -}} {{- if trim . " \t" -}} {{- $n := sub (len .) (len (strings.TrimLeft " \t" .)) -}} {{- if or (lt $indent 0) (lt $n $indent) -}}{{- $indent = $n -}}{{- end -}} {{- end -}} {{- end -}} {{- if gt $indent 0 -}} {{- $inner = replaceRE (printf "(?m)^[ \t]{%d}" $indent) "" $inner -}} {{- end -}} {{- partial "notes.html" (dict "type" $type "size" (.Get "size") "text" ($inner | markdownify)) -}}
