{{- /* Turns an `editions:` list of ids into display names for a badge row: {{ range partial "releases/edition-names.html" (dict "editions" .Params.editions "where" .RelPermalink) }} {{ . }} {{ end }} New entries author V6 edition ids from data/pulumi_pricing.yaml. Entries from before September 15, 2026 can use the V5 ids so their historical availability does not change when V6 launches. An unknown id fails the build, so a badge can never silently render a slug or nothing at all. Returns an empty slice when there are no editions, so callers can range unconditionally. */ -}} {{- $out := slice -}} {{- with .editions -}} {{- $px := partialCached "pricing/data.html" "pricing-data" "pricing-data" -}} {{- $ids := slice -}} {{- range $px.editions -}}{{- $ids = $ids | append .id -}}{{- end -}} {{- $legacyNames := dict "individual" "Individual" "team" "Team" "enterprise" "Enterprise" "business-critical" "Business Critical" -}} {{- range . -}} {{- $id := printf "%v" . -}} {{- $name := index $px.names $id -}} {{- if not $name -}}{{- $name = index $legacyNames $id -}}{{- end -}} {{- if not $name -}} {{- errorf "editions: %q is not a current or historical edition id (in %s). Current ids: %s." $id $.where (delimit $ids ", ") -}} {{- else -}} {{- $out = $out | append $name -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
