Parameters: - breadcrumb_label: Right-side breadcrumb segment (e.g., "May 2026"). - heading: Card heading. Rendered as raw HTML, so inline tags like
can be used to control line breaks. - description: Body copy (markdown). - hero_image: Image path. - hero_image_alt: Alt text for the hero image. - hero_image_max_width: Optional max-width (in px) for the image, centered horizontally and flush with the bottom of the card. Image goes full-width when unset. - hero_image_max_height: Optional max-height (in px). Sizes the image by height instead of width, letting the width follow the artwork's own aspect ratio. Use this for wide hero art, which reads too small when capped by width. Takes precedence over hero_image_max_width. */}} {{ $breadcrumbLabel := .breadcrumb_label }} {{ $heading := .heading }} {{ $description := .description }} {{ $heroImage := .hero_image }} {{ $heroImageAlt := .hero_image_alt | default $heading }} {{ $heroImageMaxWidth := .hero_image_max_width }} {{ $heroImageMaxHeight := .hero_image_max_height }} {{/* Two sizing modes. By default the wrapper caps the image's width and the image fills it. When hero_image_max_height is set we instead cap the image's height and let its width follow the artwork's aspect ratio — wide art ends up much larger on the page for the same visual weight. */}} {{ $maxWidthAttr := "" }} {{ $imgClass := "w-full h-auto block" }} {{ $imgStyle := "" }} {{ if $heroImageMaxHeight }} {{ $imgClass = "w-auto h-auto max-w-full block mx-auto" }} {{/* safeCSS: the value lands in a style attribute, which Hugo otherwise sanitizes to ZgotmplZ. */}} {{ $imgStyle = printf "max-height: %vpx;" $heroImageMaxHeight | safeCSS }} {{ else if $heroImageMaxWidth }} {{ $maxWidthAttr = printf ` style="max-width: %vpx;"` $heroImageMaxWidth }} {{ end }} {{ $heroImageExists := false }} {{ with $heroImage }} {{ $fpSrc := strings.TrimPrefix "/" . }} {{ if resources.Get (printf "fingerprinted/%s" $fpSrc) }} {{ $heroImageExists = true }} {{ end }} {{ end }}
