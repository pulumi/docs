{{/* Resolves a taxonomy term page's real display name for structured data, matching whatever name the page's own H1 uses. Hugo's default taxonomy-term title is generated by title-casing the raw slug (e.g. "azure-top-5" -> "Azure-Top-5", "best-practices" -> "Best-Practices"), which is what a term page falls through to when it carries no title_tag front matter — true of every blog category/series/tag/author archive. Each of those term kinds already resolves its own H1 from a data file or a slug-humanize rule (see layouts/taxonomy/*.html); this partial reuses those exact same lookups so the two can never drift apart. Non-term pages, and any term kind not special-cased below (currently only "industry"), get back the caller's own default unchanged. Params: page (the page context) · default (fallback string). */}} {{ $page := .page }} {{ $name := .default }} {{ if eq $page.Kind "term" }} {{ $term := $page.Data.Term }} {{ $singular := $page.Data.Singular }} {{ if eq $singular "author" }} {{/* See layouts/taxonomy/author.html: real name from the team roster. */}} {{ with index site.Data.team.team $term }}{{ $name = .name }}{{ end }} {{ else if eq $singular "category" }} {{/* See layouts/taxonomy/category.html: name from data/blog_categories.yaml. */}} {{ range site.Data.blog_categories.categories }} {{ if eq .id $term }}{{ $name = .name }}{{ end }} {{ end }} {{ else if eq $singular "series" }} {{/* See layouts/taxonomy/series.html + blog/series-page.html: title from data/blog_series.yml when the slug has an entry there, else the same humanized fallback label the plain-listing branch renders. */}} {{ $found := false }} {{ with site.Data.blog_series }} {{ range .series }} {{ if eq .slug $term }}{{ $name = .title }}{{ $found = true }}{{ end }} {{ end }} {{ end }} {{ if not $found }}{{ $name = printf "Series: %s" ($term | humanize | lower) }}{{ end }} {{ else if eq $singular "tag" }} {{/* See layouts/taxonomy/tag.html: no data file, just a humanized label. */}} {{ $name = printf "Tag: %s" ($term | humanize | lower) }} {{ end }} {{ end }} {{ return $name }}
