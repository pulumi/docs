{{- /* Returns the display names of the editions a feature from data/pulumi_editions.yaml is available in, lowest first: the feature's `min_edition` and every edition above it, in `editions[]` order. Expects a dict: feature a feature id from data/pulumi_editions.yaml where a human-readable location used in build errors (page path) An unknown id is a hard build failure. Unlike cloud-availability-body.html, a feature available on the lowest edition is fine here — the caller decides whether "every edition" is worth saying. */ -}} {{- $id := printf "%v" .feature -}} {{- $px := partialCached "pricing/data.html" "pricing-data" "pricing-data" -}} {{- $f := index $px.features $id -}} {{- if not $f -}} {{- errorf "%q is not a feature in data/pulumi_editions.yaml (in %s). Known ids: %s" $id .where (delimit $px.ids ", ") -}} {{- end -}} {{- $min := $f.min_edition -}} {{- if not $min -}} {{- errorf "feature %q in data/pulumi_editions.yaml isn't available on any edition (in %s), so there are no editions to list." $id .where -}} {{- end -}} {{- $names := slice -}} {{- $seen := false -}} {{- range $e := $px.editions -}} {{- if eq $e.id $min -}}{{- $seen = true -}}{{- end -}} {{- if $seen -}}{{- $names = $names | append $e.name -}}{{- end -}} {{- end -}} {{- return $names -}}
