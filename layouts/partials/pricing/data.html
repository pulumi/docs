{{- /* Indexes data/pulumi_editions.yaml once per build and returns the shape every consumer wants. That file is generated by pulumi/marketing-web from apps/www/src/data/pricing/, which is the source of truth for the editions and feature availability, and its sync PRs keep the copy here current. Never edit it here. Call this through partialCached with the constant key "pricing-data" — it is page-independent, and keying it by anything else would rebuild the index per page: {{ $px := partialCached "pricing/data.html" . "pricing-data" }} Returns a dict: editions the editions list ({id, name}), verbatim and in order order edition id -> ordinal (0-based) names edition id -> display name ("enterprise" -> "Enterprise") first id of the lowest edition (the one that gates nothing) features feature id -> {id, name, min_edition, hidden}. min_edition is the edition a reader has to buy; the feature is available there and in every edition after it. ids every feature id, sorted, for error messages The data is validated where it's authored, in marketing-web. The checks here only guard against a sync that breaks the shape this partial relies on. */ -}} {{- $where := "data/pulumi_editions.yaml" -}} {{- $px := site.Data.pulumi_editions -}} {{- if not $px -}} {{- errorf "%s is missing or empty. It is generated by pulumi/marketing-web; restore it from that repo's latest sync PR." $where -}} {{- end -}} {{- $editions := $px.editions -}} {{- $order := dict -}} {{- $names := dict -}} {{- $editionIds := slice -}} {{- range $i, $e := $editions -}} {{- $order = merge $order (dict $e.id $i) -}} {{- $names = merge $names (dict $e.id $e.name) -}} {{- $editionIds = $editionIds | append $e.id -}} {{- end -}} {{- $features := dict -}} {{- $ids := slice -}} {{- range $f := $px.features -}} {{- if not (in $editionIds $f.min_edition) -}} {{- errorf "%s: feature %q has min_edition: %q, which is not an edition id (%s)." $where $f.id $f.min_edition (delimit $editionIds ", ") -}} {{- end -}} {{- $features = merge $features (dict $f.id $f) -}} {{- $ids = $ids | append $f.id -}} {{- end -}} {{- return (dict "editions" $editions "order" $order "names" $names "first" (index $editionIds 0) "features" $features "ids" (sort $ids)) -}}
