{{- $where := "data/pulumi_editions.yaml" -}} {{- $px := site.Data.pulumi_editions -}} {{- if not $px -}} {{- errorf "%s is missing or empty. It is generated by pulumi/marketing-web; restore it from that repo's latest sync PR." $where -}} {{- end -}} {{- $editions := $px.editions -}} {{- $order := dict -}} {{- $names := dict -}} {{- $editionIds := slice -}} {{- range $i, $e := $editions -}} {{- $order = merge $order (dict $e.id $i) -}} {{- $names = merge $names (dict $e.id $e.name) -}} {{- $editionIds = $editionIds | append $e.id -}} {{- end -}} {{- $features := dict -}} {{- $ids := slice -}} {{- range $f := $px.features -}} {{- if not (in $editionIds $f.min_edition) -}} {{- errorf "%s: feature %q has min_edition: %q, which is not an edition id (%s)." $where $f.id $f.min_edition (delimit $editionIds ", ") -}} {{- end -}} {{- $features = merge $features (dict $f.id $f) -}} {{- $ids = $ids | append $f.id -}} {{- end -}} {{- return (dict "editions" $editions "order" $order "names" $names "first" (index $editionIds 0) "features" $features "ids" (sort $ids)) -}}
